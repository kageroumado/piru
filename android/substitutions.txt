# Regex substitutions applied to every staged Swift file, after the patch series.
# Format: <python regex> TAB <replacement>. Lines starting with # are comments.
#
# Apple modules are replaced by rewriting their imports rather than by a module of the same
# name: Skip's own packages probe `canImport(os)`, `canImport(OSLog)` and the like, and a
# stand-in carrying the Apple name makes them take their Darwin branches.

# Logger comes from SkipFuse (Skip's AndroidLogging); the privacy interpolation,
# OSAllocatedUnfairLock and signposts from Android/OSCompat.swift.
^([ \t]*)import (os|OSLog)$	\1import SkipFuse
# semantic: whether an @Observable model's changes re-render the views reading it. The macro
# names `Observation.ObservationRegistrar`, which reaches Compose only as SkipAndroidBridge's
# struct of that name, visible where SkipFuse is imported; a file that reaches the macro some
# other way (Observation alone, or SwiftData through PortableData) gets the plain registrar, and
# its views never update. So every file declaring an @Observable type imports SkipFuse.
\A(?=[\s\S]*?^[ \t]*(?:@\w+[ \t]+)*@Observable\b)	import SkipFuse\n
^([ \t]*)import SwiftData$	\1import PortableData
^([ \t]*)import CryptoKit$	\1import Crypto
# CLLocationCoordinate2D comes from Android/AndroidPlatformTypes.swift.
^import CoreLocation$

# Skip Fuse bridges every view and every property-wrapped view property to Compose, and it
# cannot bridge a private one. No private type name in the app collides with another
# type's, so lifting them to internal changes no meaning.
^(\s*)((?:nonisolated\s+|final\s+)*)(?:private|fileprivate)\s+((?:nonisolated\s+|final\s+)*(?:struct|enum|class|actor)\s)	\1\2\3
^(\s*@[^\n]*?)\b(?:private|fileprivate)\s+((?:static\s+)?(?:var|let)\b)	\1\2
^(\s*@(?!ViewBuilder\b)\w+(?:\(.*\))?[ \t]*\n\s*)(?:private|fileprivate)\s+((?:static\s+)?(?:var|let)\b)	\1\2

# semantic: every setting lives in UserDefaults.standard. Android has no app group, and Skip's
# UserDefaults(suiteName:) is a second SharedPreferences file, so the Settings UI and the code
# reading the group suite would disagree. The .standard and app-group key sets are disjoint, so
# one store holds both. @AppStorage drops its suite (a store: argument beside @Bindable fails to
# compile); every other construction reads the twin, which is .standard.
@AppStorage\(([^,()]+), store: UserDefaults\(suiteName: [^()]+\)\)	@AppStorage(\1)
\bUserDefaults\(suiteName: [^()]+\)	UserDefaults.androidAppGroup
